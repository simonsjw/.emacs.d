** Required environmental variables

Export these from the login shell (=~/.bashrc= / =~/.bash_profile=)
before =emacs-daemon-wrapper= starts.  Do not put API keys in this
file; they belong in ~$ENV_FILE~.  Narrative for how Emacs finds each
variable is in =docs/os-integrations.org=.

| Variable | Purpose |
|----------+---------|
| ~MY_NAME~ | Namespace under =var/= and =etc/= (default ~INFODYNAMICS~) |
| ~USER_EMACS_DIRECTORY~ | Config root (this tree) |
| ~ENV_FILE~ | Extra ~KEY=value~ dump loaded at startup |
| ~BIB_HOME~ | Default bibliography for citar |
| ~REPO_LIST~ | CSV of known git repos |
| ~PROJECT_PATHS~ | Project search roots: comma-separated ~dir:depth~ entries |
| ~CONDA_PREFIX~ / ~CONDA_DEFAULT_ENV~ / ~WORKON_HOME~ | Python / Conda environments |
| ~JUPYTER_WORKING_DIRECTORY~ / ~JUPYTER_ANACONDA_ENVIRONMENT~ | Jupyter defaults |
| ~QHOME~ / ~QSPACE~ / ~Q_MAIN_PROJECT_PATH~ | kdb/Q, if used |

Example (SYSTEM76 / INFODYNAMICS-shaped paths):

#+begin_src bash :tangle no
export MY_NAME=INFODYNAMICS
export USER_EMACS_DIRECTORY="/home/simon/.emacs.d"
export ENV_FILE="/home/simon/Me/secrets/serviceEnv.txt"
export BIB_HOME="/home/simon/sync/primary/dotfiles/latex/templates/bibFile/refs.bib"
export REPO_LIST="/home/simon/sync/primary/dotfiles/git/repoList.csv"
# Comma-separated dir:depth search roots.
export PROJECT_PATHS="/home/simon/Downloads/github:5,/home/simon/sync/primary/Adventures:5,/home/simon/sync/primary/dotfiles:5"
export CONDA_PREFIX="/home/simon/anaconda3"
export CONDA_DEFAULT_ENV="base"
export WORKON_HOME="/home/simon/anaconda3/envs/"
export JUPYTER_WORKING_DIRECTORY="/home/simon"
export JUPYTER_ANACONDA_ENVIRONMENT="gpt4"
#+end_src

Rename ~PROJECTILE_PATHS~ to ~PROJECT_PATHS~ in =~/.bashrc= (or
=~/.bash_profile=) and in ~$ENV_FILE~ on each machine.  Do that out of
band; do not commit those files here.  Lisp does not read this
variable yet.

** ImageMagick PDF policy

Allow ImageMagick to produce PDFs so Org/LaTeX preview works.  The
policy file is under =/etc/ImageMagick-6/= or =/etc/ImageMagick/=.

Change this:
#+begin_src xml :tangle no

<policy domain="coder" rights="none" pattern="PDF" />

#+end_src
to this:
#+begin_src xml :tangle no

<policy domain="coder" rights="read|write" pattern="PDF" />

#+end_src

#+begin_src emacs-lisp :tangle ./early-init.el

;;; Commentary:
;; Tangled from conf.org.  Work that is done before Emacs can be
;; initialised.  All paths + package management + GC + native-comp
;; now live here.

;;; Code:

;; === Base file paths + MY_NAME support ===
(setq user-emacs-directory
      (or (getenv "USER_EMACS_DIRECTORY")
          (expand-file-name "~/.emacs.d/")))
   

(defvar envvar/SYSTEM_NAME
  (or (getenv "MY_NAME") "INFODYNAMICS")
  "The name of the system on which we are currently running Emacs.")

;; === NO-LITTERING paths + PACKAGE SETUP (all early) ===

(defvar no-littering-var-directory
  (expand-file-name (concat "var/" envvar/SYSTEM_NAME "/")
                    user-emacs-directory))

(defvar no-littering-etc-directory
  (expand-file-name (concat "etc/" envvar/SYSTEM_NAME "/")
                    user-emacs-directory))

;; Force package.el to use our clean location from the VERY beginning
;; (Directory where ELPA/MELPA packages are installed.)
(setq package-user-dir
      (expand-file-name "elpa/" no-littering-var-directory))

;; Redirect native-comp cache
(setq my-paths/eln-cache
      (expand-file-name "eln-cache/" no-littering-etc-directory))
(when (and (fboundp 'startup-redirect-eln-cache)
           (boundp 'my-paths/eln-cache))
  (message "eln-cache file path set.")
  (startup-redirect-eln-cache my-paths/eln-cache))
  
  ;; Load path configuration (single source of truth)
(load-file (expand-file-name "custom-modules/path-support.el" user-emacs-directory))

;;;; UI configuration
(add-to-list 'default-frame-alist '(background-color . "#1B152D"))
(add-to-list 'default-frame-alist '(foreground-color . "#EEEEEC"))
(push '(tool-bar-lines . 0) default-frame-alist)
(push '(menu-bar-lines . 0) default-frame-alist)

(setq gc-cons-threshold (* 50 1000 1000))
(setq inhibit-compacting-font-caches t)
(setq comp-speed 1)
(setq load-prefer-newer t)
(setq package-native-compile t)
(setq native-comp-async-report-warnings-errors nil)

**** Load environmental variables

#+begin_src emacs-lisp :tangle ./init.el

;;; Load current environmental variables
  (defun load-environment-variables-from-file (file-path)
    "Load environment variables from a file specified by FILE-PATH."
    (with-temp-buffer
      (insert-file-contents file-path)
      (goto-char (point-min))
      (while (not (eobp))
        (let ((line
               (buffer-substring-no-properties
                (line-beginning-position) (line-end-position))))
          (when (string-match "\\([^=]+\\)=\\(.*\\)" line)
            (let ((key (match-string 1 line))
                  (value (match-string 2 line)))
              (setenv key value))))
        (forward-line 1))))

  (let ((envFile  (getenv "ENV_FILE")))
    (log/debug :fn 'init
               :msg "Load environmental variables from file."
               :obj envFile)
    (load-environment-variables-from-file envFile))

#+end_src

Show the loading time via a the emacs-startup-hook.

#+begin_src emacs-lisp :tangle ./init.el


  ;; ---------------------------------------------
  ;; All config and support files are now loaded.

  ;; Profile emacs startup
  (defun my-startup/display-startup-time ()
    "Display the startup time after Emacs is fully initialized."
    (log/info :fn 'init
              :msg (format "Emacs loaded in %s." (emacs-init-time))
              :obj t))

  (add-hook
   'emacs-startup-hook #'my-startup/display-startup-time)

  (setq inhibit-startup-screen nil                                                 ; stop the default splash screen
        inhibit-startup-message nil
        inhibit-startup-echo-area-message nil)

  (setq initial-scratch-message nil)                                             ; remove the message in the scratch buffer.


  (recentf-mode 1)
  (savehist-mode 1)

  (provide 'init)
    ;;; init.el ends here
